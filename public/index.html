Consulta Benchmarkeada:
SELECT st.genre, st.artist_name, count() AS fav_count, avg(st.popularity) AS avg_pop
FROM favoritas f
JOIN spotify_tracks st ON f.track_id = st.track_id
JOIN lista_usuarios lu ON f.dni = lu.dni
JOIN lista_direcciones ld ON lu.id_dirección = ld.id
GROUP BY st.genre, st.artist_name
ORDER BY fav_count DESC
LIMIT 20
